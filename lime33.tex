\begin{abstract}
\noindent
เอกสารฉบับนี้เป็นคู่มือฉบับสมบูรณ์สำหรับระบบประเมินและอธิบายผลความเสี่ยงเครดิต (Explainable AI: LIME \& SHAP) โดยครอบคลุม: 
(1) ตารางเปรียบเทียบความแตกต่างระหว่างโปรแกรมเดิมและโปรแกรมใหม่ 
(2) โครงสร้างและรายละเอียดฟิลด์ของตารางใหม่ \ft{ibond_33features_panel_941firm} (95 ฟิลด์อย่างละเอียด) 
(3) บัญชีรายชื่อบริษัททั้ง 941 บริษัท พร้อมช่วงเวลาเริ่มต้น--สิ้นสุด จำนวนงวดเดือน เปอร์เซ็นต์ความครบถ้วนของฟีเจอร์ และสถานะประวัติวิกฤต
(4) แผนผังขั้นตอนอัลกอริทึม (Algorithm Flowchart) 
(5) แกลเลอรีภาพผลการวิเคราะห์ในรูปแบบไฟล์ภาพ JPG พร้อมสรุปผลการประเมินความเสี่ยงของ 20 บริษัทตัวอย่างในตลาดทุนไทย และ
(6) ซอร์สโค้ดฉบับเต็มของ \ft{data_adapter.py} และ \ft{lime33_adapter_panel.py}
\end{abstract}

\section{การประเมินความแม่นยำ 3 ชั้น: วิธี A, B, C (Three-Layer Review Queue)}

\subsection{1. นิยามและหลักการของทั้ง 3 วิธี}

การใช้เกณฑ์ตัดความน่าจะเป็น (Single PD Threshold) เพียงอย่างเดียว มักเกิดช่องโหว่ในการตรวจจับ (Failure Modes) ด้วย 3 สาเหตุที่แตกต่างกัน ระบบเตือนภัยล่วงหน้าจึงกำหนดกลไกการคัดกรอง 3 ชั้น (Three-Layer Multi-Stage Framework) เพื่อประสานการทำงานร่วมกัน:

\begin{enumerate}[leftmargin=1.5em]
  \item \textbf{วิธี A: เกณฑ์ระดับความเสี่ยง (Level Rule -- PD $\ge$ Threshold)}:
  คัดกรองบริษัทที่มีค่าความน่าจะเป็นผิดนัดชำระหนี้สะสมอยู่ในกลุ่มสูงสุดของตลาด ($\text{PD} \ge \text{Threshold}_{5\%}$) ซึ่งดักจับบริษัทที่ความเสี่ยงปรากฏชัดเจนบนแบบจำลองแล้ว เช่น \ft{A}, \ft{PF}, \ft{TPOLY}, \ft{STARK}, \ft{ALL}
  
  \item \textbf{วิธี B: เกณฑ์ความเปราะบางต่อการเคลื่อนไหวของปัจจัย (Fragility / Shock Sensitivity Rule)}:
  คำนวณย้อนกลับจาก Shock Ladder ว่าหากตัวแปรสำคัญตัวใดตัวหนึ่ง (เช่น \ft{ROA}, \ft{TDTA}, \ft{Amihud}) ขยับเพียงตัวเดียว บริษัทจะต้องขยับตัวแปรไปไกลกี่เท่าของส่วนเบี่ยงเบนมาตรฐาน ($\le 1.0\,\text{SD}$) จึงจะถูกผลักข้ามเส้นเตือนภัย เพื่อดักจับบริษัทที่ค่า PD ปัจจุบันยังไม่ข้ามเส้น แต่มีความเปราะบางสูงมาก เช่น \ft{SQ}, \ft{PRIME}, \ft{EP}
  
  \item \textbf{วิธี C: เกณฑ์ความเสี่ยงแฝงทางงบดุล (Masked Distress / Fundamental Solvency Rule)}:
  คัดกรองบริษัทที่มีกำไรสะสมต่อสินทรัพย์ (\ft{REtoTA}) และอัตราส่วนเงินสด (\ft{CashRatio}) อยู่ในกลุ่มต่ำสุด $10\%$ ล่างสุดของตลาด (Bottom Decile) โดยตัดตัวแปรสภาพคล่องการเทรดหุ้นออก เพื่อป้องกันกรณีที่หุ้นยังมีการซื้อขายคึกคักทำให้ค่าความเสี่ยงในแบบจำลองหลักถูกกดต่ำเกินจริง เช่น \ft{JCK}, \ft{CHO}
\end{enumerate}

\subsection{1. ซอร์สโค้ด \ft{data_adapter.py}}
\noindent
ไฟล์โมดูล \ft{data_adapter.py} ทำหน้าที่เป็น Data Adapter ในการเชื่อมต่อตาราง SQLite และจัดการ Feature Engineering สำหรับทั้ง 2 ชุดข้อมูล:

\begin{lstlisting}[language=Python]
# -*- coding: utf-8 -*-
"""
data_adapter.py -- Unified Data Adapter for Thai Bond / Corporate Credit Panels.

Provides a unified interface to load either:
  [1] Legacy Panel: `ibond_33features_panel` (16,986 rows, 219 issuers)
  [2] New 941-firm Panel: `ibond_33features_panel_941firm` (187,007 rows, 941 firms)
"""
from __future__ import annotations

import os
import sqlite3
import numpy as np
import pandas as pd
from typing import Tuple, List, Optional, Dict, Any

BOND_33_FEATURES = [
    "amihud_monthly", "adj_illiq_kz", "percent_zero_days", "zero_days", "n_days",
    "ROA", "ROE", "DE", "CurrentRatio", "QuickRatio", "CashRatio",
    "EBITtoTA", "REtoTA", "WorkingCapitaltoTA", "TDTA", "LTDtoTA", "STDtoTA",
    "cf_Interestcoverageratio", "acc_DebtServiceCoverageRatio",
    "lnTotalAssets", "lnAge",
    "Policyrate", "GDPgrowth", "UnemploymentratemodeledILOe",
    "ESGScore", "GovernancePillarScore", "EnvironmentalPillarScore",
    "SocialPillarScore", "IndependentBoardMembers", "AverageBoardTenure",
]

class DataAdapter:
    DATASET_CHOICES = {
        1: {
            "name": "ของเก่า: ibond_33features_panel",
            "table": "ibond_33features_panel",
            "description": "16,986 issuer-months, 219 corporate bond issuers (2007-2026)"
        },
        2: {
            "name": "ข้อมูลใหม่: ibond_33features_panel_941firm",
            "table": "ibond_33features_panel_941firm",
            "description": "187,007 firm-months, 941 firms across Thai capital market (2007-2026)"
        }
    }

    @staticmethod
    def resolve_db_path(custom_path: Optional[str] = None) -> str:
        if custom_path and os.path.exists(custom_path):
            return custom_path
        
        candidates = [
            os.path.join(os.path.dirname(os.path.abspath(__file__)), "cmdf_credit.db"),
            os.path.join(os.path.dirname(os.path.abspath(__file__)), "bond_financials.db"),
            r"d:\tadgan_gaf\cmdf_credit_app\cmdf_credit.db",
            r"d:\tadgan_gaf\cmdf_credit_app\thaibma\dataset\datasets_bond\cmdf_credit.db",
            r"d:\tadgan_gaf\cmdf_credit_app\thaibma\dataset\datasets_bond\bond_financials.db",
        ]
        for p in candidates:
            if os.path.exists(p):
                return p
        raise FileNotFoundError("Could not locate cmdf_credit.db or bond_financials.db.")

    @classmethod
    def show_menu(cls) -> int:
        print("=" * 80)
        print("เลือกชุดข้อมูล (Dataset Selection Menu):")
        for key, meta in cls.DATASET_CHOICES.items():
            print(f"  [{key}] {meta['name']} ({meta['description']})")
        print("=" * 80)
        while True:
            try:
                choice_str = input("กรุณาเลือก (1 หรือ 2) [default: 1]: ").strip()
                if choice_str == "":
                    return 1
                choice = int(choice_str)
                if choice in cls.DATASET_CHOICES:
                    return choice
                print("ตัวเลือกไม่ถูกต้อง กรุณาระบุ 1 หรือ 2")
            except (EOFError, KeyboardInterrupt):
                print("\nDefaulting to 1.")
                return 1
            except ValueError:
                print("กรุณาใส่ตัวเลข 1 หรือ 2")

    @classmethod
    def load(cls, choice: int = 1, db_path: Optional[str] = None, verbose: bool = True) -> Tuple[pd.DataFrame, pd.DataFrame, pd.Series, List[str]]:
        if choice not in cls.DATASET_CHOICES:
            raise ValueError(f"Choice must be 1 or 2, got {choice}")
            
        resolved_db = cls.resolve_db_path(db_path)
        table_name = cls.DATASET_CHOICES[choice]["table"]
        
        if verbose:
            print(f"Connecting to DB: {resolved_db}")
            print(f"Loading Table: '{table_name}' ({cls.DATASET_CHOICES[choice]['name']})")
            
        con = sqlite3.connect(resolved_db)
        try:
            cur = con.cursor()
            cur.execute("SELECT name FROM sqlite_master WHERE type='table';")
            tables = [r[0] for r in cur.fetchall()]
            
            if table_name not in tables:
                csv_new = os.path.join(os.path.dirname(os.path.abspath(__file__)), "merged_panel_monthly_941firm_v20260823.csv")
                if choice == 2 and os.path.exists(csv_new):
                    if verbose:
                        print(f"Table '{table_name}' not found, falling back to CSV: {csv_new}")
                    panel = pd.read_csv(csv_new, low_memory=False)
                else:
                    raise RuntimeError(f"Table '{table_name}' not found in {resolved_db}")
            else:
                panel = pd.read_sql(f"SELECT * FROM {table_name}", con)
                
            dflt = pd.read_sql("SELECT * FROM ibond_default_payment", con) if "ibond_default_payment" in tables else pd.DataFrame()
        finally:
            con.close()
            
        if "symbol" in panel.columns and "issuer_code" not in panel.columns:
            panel["issuer_code"] = panel["symbol"].astype(str)
        if "name" in panel.columns and "issuer_name" not in panel.columns:
            panel["issuer_name"] = panel["name"].astype(str)
            
        panel["month_dt"] = pd.to_datetime(panel["month"], errors="coerce")
        
        if choice == 1:
            y = pd.Series(0, index=panel.index, dtype=int)
            if not dflt.empty:
                d = dflt.copy()
                d["issuer_code"] = d["symbol"].astype(str).str.extract(r"^([A-Z]+)")[0]
                d["payment_date"] = pd.to_datetime(d["payment_date"], errors="coerce")
                first = (d.dropna(subset=["payment_date"])
                         .groupby("issuer_code")["payment_date"].min())
                ev = panel["issuer_code"].map(first)
                gap = ((ev.dt.year - panel["month_dt"].dt.year) * 12
                       + (ev.dt.month - panel["month_dt"].dt.month))
                y = ((gap >= 0) & (gap <= 3)).fillna(False).astype(int)
                panel["event_date"] = ev
            elif "event" in panel.columns:
                y = panel["event"].fillna(0).astype(int)
            else:
                y = (panel["DE"] > panel["DE"].quantile(0.90)).astype(int)
        else:
            if "y_pre3m" in panel.columns:
                y = panel["y_pre3m"].fillna(0).astype(int)
            elif "d_DP_RS" in panel.columns:
                y = panel["d_DP_RS"].fillna(0).astype(int)
            elif "d_Default_Payment" in panel.columns:
                y = panel["d_Default_Payment"].fillna(0).astype(int)
            else:
                y = (panel["DE"] > panel["DE"].quantile(0.90)).astype(int)
                
        panel["y"] = y
        
        cols = [c for c in BOND_33_FEATURES if c in panel.columns]
        X = panel[cols].apply(pd.to_numeric, errors="coerce")
        X = X.fillna(X.median(numeric_only=True)).fillna(0.0)

            print(f"  Distress/Default events: {int(y.sum()):,} months ({y.mean()*100:.3f}%) from {panel.loc[y == 1, 'issuer_code'].nunique():,} firms")
            
        return panel, X, y, cols

"""

side by side, supporting both Legacy and New 941-Firm SQLite tables with JPG export.

  - Menu selection: [1] Legacy (ibond_33features_panel) or [2] New (ibond_33features_panel_941firm)

"""

def load_state(choice: int = 1, workload: float = DEFAULT_WORKLOAD, force: bool = False):
    cache_key = (choice, workload)
    if not force and cache_key in _STATE_CACHE:
        return _STATE_CACHE[cache_key]

    panel, X, y, cols = DataAdapter.load(choice=choice, verbose=True)
    panel = panel.reset_index(drop=True)
    A = X.to_numpy(float)
    yv = y.to_numpy(int)
    groups = panel["issuer_code"].astype(str).to_numpy()

    # StratifiedGroupKFold out-of-fold models
    fold_of = np.full(len(A), -1, int)
    fold_models = {}
    
    n_pos = int(yv.sum())
    n_splits = min(5, max(2, n_pos)) if n_pos >= 2 else 2
    
    cv = StratifiedGroupKFold(n_splits=n_splits, shuffle=True, random_state=SEED)
    for k, (tr, te) in enumerate(cv.split(A, yv, groups)):
        if yv[tr].sum() < 1:
            continue
        sck = StandardScaler().fit(A[tr])
        if CatBoostClassifier is not None:
            mk = CatBoostClassifier(iterations=300, depth=3, learning_rate=0.05,
                                    l2_leaf_reg=3.0, auto_class_weights="Balanced",
                                    random_seed=SEED, verbose=0,
                                    allow_writing_files=False).fit(sck.transform(A[tr]), yv[tr])
        else:
            from sklearn.ensemble import RandomForestClassifier
            mk = RandomForestClassifier(n_estimators=200, max_depth=6, class_weight="balanced",
                                        random_state=SEED).fit(sck.transform(A[tr]), yv[tr])
        fold_models[k] = (sck, mk)
        fold_of[te] = k

    oof = np.full(len(A), np.nan)
    for k, (sck, mk) in fold_models.items():
        m = fold_of == k
        oof[m] = mk.predict_proba(sck.transform(A[m]))[:, 1]
    
    nan_mask = np.isnan(oof)
    if nan_mask.any() and len(fold_models) > 0:
        sck_def, mk_def = list(fold_models.values())[0]
        oof[nan_mask] = mk_def.predict_proba(sck_def.transform(A[nan_mask]))[:, 1]

    last_rows = (panel.assign(_i=np.arange(len(panel)))
                 .sort_values("month_dt").groupby("issuer_code").tail(1)["_i"]
                 .to_numpy())
    cross = oof[last_rows]
    cross = cross[np.isfinite(cross)]
    thr = float(np.quantile(cross, 1.0 - workload)) if len(cross) > 0 else 0.05

    tab = pd.DataFrame({
        "issuer": panel.loc[last_rows, "issuer_code"].to_numpy(),
        "month": panel.loc[last_rows, "month"].to_numpy(),
        "pd": oof[last_rows],
        "status": np.where(oof[last_rows] >= thr, "HIGH RISK", "NORMAL")
    }).sort_values("pd", ascending=False).reset_index(drop=True)

    sc_full = StandardScaler().fit(A)
    if CatBoostClassifier is not None:
        cb_full = CatBoostClassifier(iterations=300, depth=3, learning_rate=0.05,
                                     l2_leaf_reg=3.0, auto_class_weights="Balanced",
                                     random_seed=SEED, verbose=0,
                                     allow_writing_files=False).fit(sc_full.transform(A), yv)
    else:
        from sklearn.ensemble import RandomForestClassifier
        cb_full = RandomForestClassifier(n_estimators=200, max_depth=6, class_weight="balanced",
                                         random_state=SEED).fit(sc_full.transform(A), yv)

    state = {
        "panel": panel, "X": X, "y": y, "cols": cols, "A": A, "yv": yv,
        "fold_of": fold_of, "fold_models": fold_models, "oof": oof,
        "thr": thr, "tab": tab, "sc": sc_full, "model": cb_full,
        "dataset_choice": choice
    }
    _STATE_CACHE[cache_key] = state
    return state

def issuer_row(S, issuer):
    p = S["panel"]
    rows = p.index[p["issuer_code"] == issuer].to_numpy()
    if len(rows) == 0:
        return None, None, None
    last = rows[np.argmax(p.loc[rows, "month_dt"].to_numpy())]
    k = int(S["fold_of"][last])
    sck, mk = S["fold_models"].get(k, (S["sc"], S["model"]))
    return last, sck, mk

def predict_fn(sck, mk):
    def f(B):
        B = np.asarray(B, float)
        if B.ndim == 1:
            B = B[None, :]
        return mk.predict_proba(sck.transform(B))
    return f

def run_lime(S, last, sck, mk, samples=N_SAMPLES, repeats=N_REPEATS, seed=SEED):
    A, cols = S["A"], S["cols"]
    x0 = A[last]
    f = predict_fn(sck, mk)

    runs = []
    for r in range(repeats):
        ex = LimeTabularExplainer(
            A, feature_names=cols, class_names=["no event", "event"],
            mode="classification", discretize_continuous=True,
            random_state=seed + r)
        e = ex.explain_instance(x0, f, num_features=len(cols), num_samples=samples)
        w = dict(e.as_map()[1])
        runs.append([w.get(j, 0.0) for j in range(len(cols))])

    M = np.array(runs)
    return pd.DataFrame(dict(
        feature=cols,
        lime=M.mean(0), lime_sd=M.std(0, ddof=1),
        lime_lo=M.min(0), lime_hi=M.max(0)
    ))

def run_shap(S, last, sck, mk, background=200, seed=SEED):
    A, cols = S["A"], S["cols"]
    rng = np.random.default_rng(seed)
    bg = A[rng.choice(len(A), size=min(background, len(A)), replace=False)]

    try:
        ex = shap.TreeExplainer(mk, shap.maskers.Independent(
            sck.transform(bg), max_samples=background),
            model_output="probability")
        sv = ex.shap_values(sck.transform(A[last][None, :]))
    except Exception:
        ex = shap.Explainer(mk.predict_proba, sck.transform(bg))
        sv = ex(sck.transform(A[last][None, :])).values[:, :, 1]

    sv = np.asarray(sv)
    if sv.ndim == 3:
        sv = sv[0, :, -1]
    elif sv.ndim == 2:
        sv = sv[0]
    base = float(np.ravel(getattr(ex, "expected_value", [0.05]))[-1]) if hasattr(ex, "expected_value") else 0.05
    return pd.DataFrame(dict(feature=cols, shap=np.ravel(sv)[:len(cols)])), base

def explain(S, issuer, samples=N_SAMPLES, repeats=N_REPEATS, seed=SEED):
    last, sck, mk = issuer_row(S, issuer)
    if last is None:
        return None
    A, cols = S["A"], S["cols"]
    x0 = A[last]
    pd_now = float(predict_fn(sck, mk)(x0[None, :])[0, 1])

    L = run_lime(S, last, sck, mk, samples, repeats, seed)
    Sh, base = run_shap(S, last, sck, mk, seed=seed)
    d = L.merge(Sh, on="feature")

    med = np.median(A, axis=0)
    sd = A.std(0, ddof=1)
    pct = np.array([100.0 * (A[:, j] < x0[j]).mean() for j in range(len(cols))])
    d["value"] = x0
    d["median"] = med
    d["pct_of_median"] = np.where(np.abs(med) > 1e-12,
                                  100.0 * (x0 - med) / np.abs(med), np.nan)
    d["sd_from_median"] = np.where(sd > 0, (x0 - med) / sd, 0.0)
    d["percentile"] = pct

    tot = np.abs(d.shap).sum()
    d["shap_share"] = 100.0 * d.shap / tot if tot > 0 else 0.0
    d["shap_pct_of_pd"] = 100.0 * d.shap / pd_now if pd_now > 0 else np.nan
    ltot = np.abs(d.lime).sum()
    d["lime_share"] = 100.0 * d.lime / ltot if ltot > 0 else 0.0

    d["dir_lime"] = np.where(d.lime > 0, "เพิ่ม", np.where(d.lime < 0, "ลด", "-"))
    d["dir_shap"] = np.where(d.shap > 0, "เพิ่ม", np.where(d.shap < 0, "ลด", "-"))
    d["agree"] = np.sign(d.lime) == np.sign(d.shap)
    d["lime_stable"] = np.sign(d.lime_lo) == np.sign(d.lime_hi)

    d = d.sort_values("shap", key=np.abs, ascending=False).reset_index(drop=True)
    return dict(issuer=issuer, row=int(last), pd_now=pd_now, base=base,
                thr=S["thr"], month=str(S["panel"].loc[last, "month"]),
                table=d, dataset_choice=S["dataset_choice"])
